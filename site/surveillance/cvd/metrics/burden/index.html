plotSvgNode = plot => {
  if (!plot) return null
  if (plot.tagName && plot.tagName.toLowerCase() === "svg") return plot
  return plot.querySelector ? plot.querySelector("svg") : null
}

plotSvgText = plot => {
  const svg = plotSvgNode(plot)
  if (!svg) return ""
  const clone = svg.cloneNode(true)
  clone.setAttribute("xmlns", "http://www.w3.org/2000/svg")
  clone.setAttribute("xmlns:xlink", "http://www.w3.org/1999/xlink")
  const computed = getComputedStyle(svg)
  clone.style.fontFamily = computed.fontFamily || "sans-serif"
  clone.style.fontSize = computed.fontSize || "15px"
  const background = document.createElementNS("http://www.w3.org/2000/svg", "rect")
  background.setAttribute("x", "0")
  background.setAttribute("y", "0")
  background.setAttribute("width", "100%")
  background.setAttribute("height", "100%")
  background.setAttribute("fill", "white")
  clone.insertBefore(background, clone.firstChild)
  return `<?xml version="1.0" encoding="UTF-8"?>\n${new XMLSerializer().serializeToString(clone)}`
}

downloadPlotBlob = (blob, filename) => {
  const url = URL.createObjectURL(blob)
  const link = document.createElement("a")

  link.click()

  setTimeout(() => URL.revokeObjectURL(url), 0)
}

plotDownloadControls = (plot, baseName) => {
  const root = document.createElement("div")
  root.className = "bnr-plot-downloads"
  const safeBase = plotDownloadBase(baseName)

  const svgButton = document.createElement("button")
  svgButton.type = "button"
  svgButton.className = "bnr-plot-download-button"
  svgButton.textContent = "Download SVG"
  svgButton.addEventListener("click", () => {
    const svgText = plotSvgText(plot)
    if (!svgText) return
    downloadPlotBlob(new Blob([svgText], {type: "image/svg+xml;charset=utf-8"}), `${safeBase}.svg`)
  })

  const pngButton = document.createElement("button")
  pngButton.type = "button"
  pngButton.className = "bnr-plot-download-button"
  pngButton.textContent = "Download PNG"
  pngButton.addEventListener("click", () => {
    const svg = plotSvgNode(plot)
    const svgText = plotSvgText(plot)
    if (!svg || !svgText) return
    const svgBlob = new Blob([svgText], {type: "image/svg+xml;charset=utf-8"})
    const svgUrl = URL.createObjectURL(svgBlob)
    const image = new Image()
    image.onload = () => {
      const viewBox = svg.viewBox && svg.viewBox.baseVal ? svg.viewBox.baseVal : null
      const width = Number(svg.getAttribute("width")) || (viewBox && viewBox.width) || 960
      const height = Number(svg.getAttribute("height")) || (viewBox && viewBox.height) || 500
      const scale = 2
      const canvas = document.createElement("canvas")
      canvas.width = Math.ceil(width * scale)
      canvas.height = Math.ceil(height * scale)
      const context = canvas.getContext("2d")
      context.scale(scale, scale)
      context.fillStyle = "white"
      context.fillRect(0, 0, width, height)
      context.drawImage(image, 0, 0, width, height)
      canvas.toBlob(blob => {
        URL.revokeObjectURL(svgUrl)
        if (blob) downloadPlotBlob(blob, `${safeBase}.png`)
      }, "image/png")
    }
    image.onerror = () => URL.revokeObjectURL(svgUrl)
    image.src = svgUrl
  })

  root.append(svgButton, pngButton)
  return root
}

  const nonAnnualCountContext = selected_indicator === "counts" && selected_period_type !== "annual"
  const annualContext = (selected_indicator === "counts" && selected_period_type === "annual") || selected_indicator === "rates"

  const completeYears = new Set(
    annualContext
      ? metrics.filter(d =>
          d.metric_id === metricId && d.period_type === "annual" && d.statistic === statistic &&
          d.event_type === selected_event_type && d.sex === "all" && d.age_group === ageGroup &&
          seriesMatches(d, "hospital") && (d.period_complete === 1 || d.period_complete === "1" || d.period_complete === true)
        ).map(d => d.period_year)
      : []
  )

  let changed = false
  Array.from(boxes).forEach((box, i) => {
    const key = optionKeys[i]
    if (!key) return
    let disable = false
    let disabledReason = ""
    if (nonAnnualCountContext && key !== "hospital") {
      disable = true
      disabledReason = "Hospital + DCO event counts are published annually only."
    } else if (annualContext && key !== "hospital" && completeYears.size > 0) {
      const publishable = metrics.filter(d =>
        d.metric_id === metricId && d.period_type === "annual" && d.statistic === statistic &&
        d.event_type === selected_event_type && d.sex === "all" && d.age_group === ageGroup &&
        completeYears.has(d.period_year) && seriesMatches(d, key) &&
        d.suppression_status === "none" && Number.isFinite(d.value)
      ).length
      disable = publishable / completeYears.size < 0.5
      if (disable) disabledReason = `${seriesLabel(key)} is not offered as a trend for ${eventLabel(selected_event_type)} because too few annual values can be published after disclosure control.`
    }
    box.disabled = disable
    const label = box.closest("label")
    if (label) {
      label.style.opacity = disable ? "0.45" : ""
      label.style.cursor = disable ? "help" : ""
      label.title = disabledReason
    }
    if (disable && box.checked) { box.checked = false; changed = true }
  })
  if (changed) {
    const remaining = Array.from(boxes).filter(box => box.checked && !box.disabled)
    if (!remaining.length && boxes[0] && !boxes[0].disabled) boxes[0].checked = true
    control.dispatchEvent(new Event("input", {bubbles: true}))
  }
}

" The published five-year rolling average is shown with lighter lines where available." : " Add the five-year rolling average using Comparator."}</p>

            seriesMatches(d, "dco_primary") ||
            seriesMatches(d, "dco_inclusive")
          )
        )
      : selected_indicator === "profile"
      ? metrics.filter(d => d.metric_id === "CVD-BURDEN-002")
      : metrics.filter(d => d.metric_id === "CVD-INCIDENCE-001")
  )

    {item: "Release date", value: release_date_label},
    {item: "Latest data period", value: latestIndicatorLabel},
    {item: "Status", value: "Published public release"},
    {item: "Protected values", value: "Protected values are shown exactly as published. The dashboard never tries to work them out."},
    {item: "Monthly comparison", value: "Optional fixed 2015–2019 historical reference; it does not change with each new release."}
  ]

  if (selected_indicator === "counts") {
    metaRows.push(
      {item: "DCO linkage uncertainty", value: "Published separately for Hospital + DCO Primary and Inclusive annual counts. Counts are registry enumeration and do not have statistical confidence intervals."}
    )
  }

  if (selected_indicator === "rates") {
    metaRows.push(
      {item: "Population source", value: "UN World Population Prospects 2024"},
      {item: "Rate unit", value: "Per 100,000 population"},
      {item: "Statistical uncertainty", value: "Published 95% confidence intervals. Hospital crude rates use exact Poisson/Garwood; hospital age-standardised rates use Fay–Feuer gamma; Hospital + DCO intervals use the corresponding conditional gamma methods based on central pseudo-counts."},
      {item: "DCO linkage uncertainty", value: "Published separately for Hospital + DCO Primary and Inclusive rates. It is not combined with the statistical confidence interval."},
      ...(selected_rate_measure === "asr"
        ? [{item: "Standard population", value: "WHO World 2000–2025"}]
        : [])
    )
  }

  const zipName = release_id ? `cvd_metrics_${release_id}.zip` : ""

  return html`<div class="bnr-data-page-stack">

          <tr><td>Release package</td><td>Complete public files for this CVD release.</td><td>${zipName ? html`<a href=${`../../../../downloads/files/metrics/cvd/releases/${zipName}`} download>Download ZIP</a>` : "—"}</td></tr>
