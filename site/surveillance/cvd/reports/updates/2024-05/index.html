releaseDescriptor = releaseId => {
  const match = String(releaseId ?? "").match(/_(\d{4})_(\d{2})$/)
  if (!match) return String(releaseId ?? "")
  const year = Number(match[1])
  const month = Number(match[2])
  const monthName = d3.utcFormat("%B")(new Date(Date.UTC(year, month - 1, 1)))
  return year + ", " + monthName
}

isPublishedNumber = row => {
  const status = String(row?.suppression_status ?? "").trim().toLowerCase()
  const value = Number(row?.display_value)
  return (status === "" || status === "none" || status === "not_applicable") && Number.isFinite(value)
}

complete = row => String(row?.period_complete) === "1"

monthKey = row =>
  String(row.period_year).padStart(4, "0") + "-" +
  String(row.period_month).padStart(2, "0")

monthSerial = key => {
  const [year, month] = key.split("-").map(Number)
  return year * 12 + month
}

monthDate = key => new Date(key + "-01T00:00:00Z")

shiftMonth = (key, offset) => {
  const [year, month] = key.split("-").map(Number)
  return d3.utcFormat("%Y-%m")(new Date(Date.UTC(year, month - 1 + offset, 1)))
}

monthLabel = key => d3.utcFormat("%B %Y")(monthDate(key))
monthShortLabel = key => d3.utcFormat("%b")(monthDate(key))

monthRangeLabel = months => {
  if (!months?.length) return ""
  const start = monthDate(months[0])
  const end = monthDate(months[months.length - 1])
  const sameYear = d3.utcFormat("%Y")(start) === d3.utcFormat("%Y")(end)
  return sameYear
    ? d3.utcFormat("%B")(start) + "\u2013" + d3.utcFormat("%B %Y")(end)
    : d3.utcFormat("%B %Y")(start) + "\u2013" + d3.utcFormat("%B %Y")(end)
}

rowReady = row => Boolean(row) && complete(row) && isPublishedNumber(row)

Rolling 12-month trend

Choose between a cumulative total and the difference from the 2015–2019 average for the same calendar month. In the difference view, the horizontal zero line is the 2015–2019 monthly average: values above zero are higher than that reference and values below zero are lower. Background lines show aligned 12-month periods from earlier years. Only complete published values are used.

referenceYears = d3.range(2015, 2020)
referencePeriodLabel = "2015–2019"

calendarMonthNumber = key => Number(String(key).slice(5, 7))

monthlyReferenceAverage = byMonth => {
  const reference = new Map()

  d3.range(1, 13).forEach(month => {
    const keys = referenceYears.map(year =>
      String(year) + "-" + String(month).padStart(2, "0")
    )
    const rows = keys.map(key => byMonth.get(key))

    // The reference is shown only when all five published reference-year values
    // are complete and available. Missing or protected values are never filled in.
    if (!rows.every(rowReady)) {
      reference.set(month, null)
      return
    }

    reference.set(
      month,
      d3.mean(rows, row => Number(row.display_value))
    )
  })

  return reference
}

rollingBlock = (byMonth, endMonth, yearsBack = 0) => {
  if (!endMonth) return null

  const shiftedEnd = shiftMonth(endMonth, -12 * yearsBack)
  const months = d3.range(-11, 1).map(offset => shiftMonth(shiftedEnd, offset))

  if (!months.every(key => rowReady(byMonth.get(key)))) return null

  const series = monthRangeLabel(months)
  const startYear = months[0].slice(0, 4)
  const endYear = months[months.length - 1].slice(0, 4)
  const seriesYearRange = startYear === endYear ? startYear : startYear + "\u2013" + endYear
  let cumulative = 0

  return months.map((month, index) => {
    const monthlyCount = Number(byMonth.get(month).display_value)
    cumulative += monthlyCount
    return {
      axisMonth: monthShortLabel(month),
      cumulative,
      monthlyCount,
      month,
      series,
      seriesYearRange,
      seriesType: yearsBack === 0 ? "current" : "historical_comparator",
      yearsBack,
      step: index + 1
    }
  })
}

differenceBlock = (byMonth, referenceByCalendarMonth, endMonth, yearsBack = 0) => {
  if (!endMonth) return null

  const shiftedEnd = shiftMonth(endMonth, -12 * yearsBack)
  const months = d3.range(-11, 1).map(offset => shiftMonth(shiftedEnd, offset))

  if (!months.every(key => rowReady(byMonth.get(key)))) return null
  if (!months.every(key => Number.isFinite(referenceByCalendarMonth.get(calendarMonthNumber(key))))) return null

  const series = monthRangeLabel(months)
  const startYear = months[0].slice(0, 4)
  const endYear = months[months.length - 1].slice(0, 4)
  const seriesYearRange = startYear === endYear ? startYear : startYear + "\u2013" + endYear

  return months.map((month, index) => {
    const monthlyCount = Number(byMonth.get(month).display_value)
    const referenceAverage = Number(referenceByCalendarMonth.get(calendarMonthNumber(month)))
    return {
      axisMonth: monthShortLabel(month),
      monthlyCount,
      referenceAverage,
      difference: monthlyCount - referenceAverage,
      month,
      series,
      seriesYearRange,
      seriesType: yearsBack === 0 ? "current" : "historical_comparator",
      yearsBack,
      step: index + 1
    }
  })
}

rollingComparators = (byMonth, endMonth) => {
  if (!endMonth || !byMonth.size) return []

  const earliestSerial = d3.min(Array.from(byMonth.keys()), monthSerial)
  const maximumYearsBack = Math.max(
    0,
    Math.floor((monthSerial(endMonth) - earliestSerial) / 12)
  )

  return d3.range(1, maximumYearsBack + 1)
    .map(yearsBack => rollingBlock(byMonth, endMonth, yearsBack))
    .filter(block => block !== null)
}

differenceComparators = (byMonth, referenceByCalendarMonth, endMonth) => {
  if (!endMonth || !byMonth.size) return []

  const earliestSerial = d3.min(Array.from(byMonth.keys()), monthSerial)
  const maximumYearsBack = Math.max(
    0,
    Math.floor((monthSerial(endMonth) - earliestSerial) / 12)
  )

  return d3.range(1, maximumYearsBack + 1)
    .map(yearsBack => differenceBlock(byMonth, referenceByCalendarMonth, endMonth, yearsBack))
    .filter(block => block !== null)
}

formatOneDecimal = value =>
  Number.isFinite(Number(value)) ? Number(value).toFixed(1) : "Unavailable"

formatSignedOneDecimal = value => {
  if (!Number.isFinite(Number(value))) return "Unavailable"
  const number = Number(value)
  if (Math.abs(number) < 0.05) return "0.0"
  return (number > 0 ? "+" : "\u2212") + Math.abs(number).toFixed(1)
}

cumulativeTrendPlot = (current, comparators, plotWidth, yDomainMaximum = null) => {
  if (!current) return null

        z: "series",
        stroke: "#607780",
        strokeOpacity: 0.18,
        strokeWidth: 1.25
      })
    )

    Plot.line(current, {
      x: "axisMonth",

      strokeWidth: 3
    })

    Plot.tip(
      interactive,
      Plot.pointerX({
        x: "axisMonth",
        y: "cumulative",
        title: "hoverTitle"
      })
    )
  )

  marks.push(
    Plot.dot(current.slice(-1), {
      x: "axisMonth",
      y: "cumulative",
      fill: "#045174",
      r: 4
    })
  )

  const yOptions = {
    grid: true,
    label: null,
    nice: true
  }
  if (
    yDomainMaximum !== null &&
    yDomainMaximum !== undefined &&
    Number.isFinite(Number(yDomainMaximum)) &&
    Number(yDomainMaximum) > 0
  ) {
    yOptions.domain = [0, Number(yDomainMaximum)]
  }

  return Plot.plot({
    width: plotWidth,
    height: 340,
    marginLeft: 62,
    marginRight: 22,
    marginBottom: 42,
    style: {
      fontFamily: "inherit",
      fontSize: "13px",
      background: "transparent"
    },
    x: {
      domain: current.map(row => row.axisMonth),
      label: null
    },
    y: yOptions,
    marks
  })
}

csvEscape = value => {
  const text = value === null || value === undefined ? "" : String(value)
  return /[",\r\n]/.test(text) ? `"${text.replaceAll('"', '""')}"` : text
}

chartCsv = rows => {
  if (!rows.length) return ""
  const headers = Object.keys(rows[0])
  return [
    headers.map(csvEscape).join(","),
    ...rows.map(row => headers.map(key => csvEscape(row[key])).join(","))
  ].join("\r\n")
}

sourceCsvWithMetadata = ({rows, sourceReleaseId, sourceSnapshot, sourceType, publishedSourceHref}) => {
  if (!rows?.length) return ""

  const metadataHeaders = [
    "report_id",
    "report_version",
    "report_date",
    "report_period",
    "source_type",
    "source_release_id",
    "source_release",
    "source_snapshot",
    "published_source_href"
  ]
  const sourceHeaders = rows.columns?.length ? rows.columns : Object.keys(rows[0])
  const headers = [...metadataHeaders, ...sourceHeaders]

  return [
    headers.map(csvEscape).join(","),
    ...rows.map(row => {
      const metadata = {
        report_id: "bnr_cvd_update_2024_05_v1",
        report_version: "v1",
        report_date: "2026-09-28",
        report_period: "May 2024",
        source_type: sourceType,
        source_release_id: sourceReleaseId,
        source_release: releaseDescriptor(sourceReleaseId),
        source_snapshot: sourceSnapshot,
        published_source_href: publishedSourceHref
      }
      return headers.map(key =>
        csvEscape(Object.prototype.hasOwnProperty.call(metadata, key) ? metadata[key] : row[key])
      ).join(",")
    })
  ].join("\r\n")
}

The full-release and chart-data CSVs are browser-generated convenience downloads from the frozen snapshots stored with this dated report. They add report and release metadata but do not replace the authoritative published release files.
